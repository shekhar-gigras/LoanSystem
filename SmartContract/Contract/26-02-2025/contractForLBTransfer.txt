pragma solidity ^0.8.0;

// Contract for managing lending and loan operations
contract LendingAndLoanManagement {
    // State variable to store the admin address
    address public Admin;

    // Constructor to set the Admin as the contract deployer
    constructor() {
        Admin = msg.sender; // Sets the contract deployer as the Admin
    }

    // Modifier to restrict access to only the Admin
    modifier onlyAdmin() {
        require(msg.sender == Admin, "Only Admin can perform this action");
        _;
    }
   

    // Struct to store common and interest details of a loan
    struct CommonAndInterestDetails {
        string emiPaymentStartDate;
        uint256 principalAmount;
        uint256 fixedInterestRate;
        uint256 monthlyPaymentAmount;
        string maturityDate;
        uint8 emiPaymentDay;
        string interestRateChangeDate;
        string lenderName;
        string borrowerName;
    }

    // Struct to store payment and late payment details of a loan
    struct PaymentAndLateDetails {
        uint8 latePaymentGracePeriod;
        uint256 lateChargePercentage;
        uint256 margin;
        uint256 currentIndex;
        uint256 maxInterestRateAtFirstChange;
        uint256 minInterestRateAtFirstChange;
        uint256 maxInterestRateAfterChange;
        uint256 minInterestRateAfterChange;
    }

    // Struct to store adjustable interest rate details of a loan
    struct AdjustableInterestRateDetails {
        string noteDate;
        string city;
        string state;
        string propertyAddress;
        string paymentLocation;
    }

    // Struct to combine all loan details
    struct Loan {
        CommonAndInterestDetails commonDetails;
        PaymentAndLateDetails paymentDetails;
        AdjustableInterestRateDetails adjustableDetails;
    }

    // Struct to store borrower details and their loans
    struct Borrower {
        string borrowerID;
        uint256[] loanIds; // Array to store the loan IDs associated with this borrower
        uint256 loanCount; // Number of loans this borrower has
        mapping(uint256 => Loan) loans; // Mapping from loan ID to loan details
    }

    // Struct to store lender details and their borrowers
    struct Lender {
        string lenderID;
        string name;
        string[] borrowerIds; // Array to store IDs of borrowers under this lender
        uint256 borrowerCount; // Number of borrowers under this lender
        mapping(string => Borrower) borrowers; // Mapping from borrower ID to borrower details
    }

    // Mapping to store lenders by their ID
    mapping(string => Lender) internal lenders;
    // Array to store all lender IDs
    string[] internal lenderIDs;

    // Events to log important actions for easier tracking
    event LenderAdded(string lenderID, string lenderName);
    event BorrowerAdded(string lenderID, string borrowerID);
    event LenderDeleted(string lenderID);
    event BorrowerDeleted(string lenderID, string borrowerID);
    event LoanDeleted(string lenderID, string borrowerID, uint256 loanID);
    event ContractFunded(address Admin, uint256 amount);
    event FundsWithdrawn(address Admin, uint256 amount);
    event LoanTransferred(
        string oldLenderID,
        string newLenderID,
        string borrowerID,
        uint256 loanID
    );
    event LoanUpdated(string lenderID, string borrowerID, uint256 loanID);
    event AdminChanged(address indexed oldAdmin, address indexed newAdmin);

     //change the Admin
    function changeAdmin(address _admin) external onlyAdmin {
        require(_admin != address(0), "New admin address cannot be zero");
        emit AdminChanged(Admin, _admin); // Log the change
        Admin = _admin;
    }
    

    // Function to add a new lender
    function addLender(string memory _lenderID, string memory _name)
        external
        onlyAdmin
    {
        // Ensure the lender does not already exist
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length == 0, "Lender already exists");

        lender.lenderID = _lenderID;
        lender.name = _name;
        lenderIDs.push(_lenderID);

        emit LenderAdded(_lenderID, _name); // Emit an event for tracking
    }

    // Function to add a new borrower and their loan details
    function addBorrower(
        string memory _borrowerID,
        string memory _lenderID,
        uint256 _loanID,
        CommonAndInterestDetails memory _commonDetails,
        PaymentAndLateDetails memory _paymentDetails,
        AdjustableInterestRateDetails memory _adjustableDetails
    ) external onlyAdmin {
        // Check that the lender exists
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length > 0, "Lender does not exist");

        // Add a new borrower if they don't exist
        Borrower storage borrower = lender.borrowers[_borrowerID];
        if (bytes(borrower.borrowerID).length == 0) {
            borrower.borrowerID = _borrowerID;
            lender.borrowerIds.push(_borrowerID);
            lender.borrowerCount++;
        }

        // Ensure the loan ID is unique for this borrower
        require(
            borrower.loans[_loanID].commonDetails.principalAmount == 0,
            "Loan ID already exists"
        );

        borrower.loanIds.push(_loanID); // Add the loan ID to the borrower's loanIds array
        borrower.loanCount++;

        // Assign loan details
        Loan storage l = borrower.loans[_loanID];
        l.commonDetails = _commonDetails;
        l.paymentDetails = _paymentDetails;
        l.adjustableDetails = _adjustableDetails;

        emit BorrowerAdded(_lenderID, _borrowerID);
    }

    // Function to update a loan's details
function updateLoan(
    string memory _lenderID,
    string memory _borrowerID,
    uint256 _loanID,
    CommonAndInterestDetails memory _newCommonDetails,
    PaymentAndLateDetails memory _newPaymentDetails,
    AdjustableInterestRateDetails memory _newAdjustableDetails
) external onlyAdmin {
    // Ensure the lender exists
    require(bytes(lenders[_lenderID].name).length > 0, "Lender does not exist");

    // Ensure the borrower exists
    require(
        bytes(lenders[_lenderID].borrowers[_borrowerID].borrowerID).length > 0,
        "Borrower does not exist"
    );

    // Ensure the loan exists
    require(
        lenders[_lenderID].borrowers[_borrowerID].loans[_loanID]
            .commonDetails
            .principalAmount > 0,
        "Loan does not exist"
    );

    // Access the loan and update the details
    Loan storage loan = lenders[_lenderID].borrowers[_borrowerID].loans[_loanID];
    loan.commonDetails = _newCommonDetails;
    loan.paymentDetails = _newPaymentDetails;
    loan.adjustableDetails = _newAdjustableDetails;

    emit LoanUpdated(_lenderID, _borrowerID, _loanID);
}


    // Function to transfer a loan from one lender to another
    function transferLoan(
        string memory _oldLenderID,
        string memory _newLenderID,
        string memory _borrowerID,
        uint256 _loanID
    ) external onlyAdmin {
        // Ensure both lenders exist
        require(
            bytes(lenders[_oldLenderID].name).length > 0,
            "Old lender does not exist"
        );
        require(
            bytes(lenders[_newLenderID].name).length > 0,
            "New lender does not exist"
        );

        // Ensure the borrower exists under the old lender
        require(
            bytes(lenders[_oldLenderID].borrowers[_borrowerID].borrowerID).length > 0,
            "Borrower does not exist"
        );

        // Ensure the loan exists under the old lender's borrower
        require(
            lenders[_oldLenderID].borrowers[_borrowerID].loans[_loanID].commonDetails.principalAmount > 0,
            "Loan does not exist"
        );

        Borrower storage oldBorrower = lenders[_oldLenderID].borrowers[_borrowerID];
        Borrower storage newBorrower = lenders[_newLenderID].borrowers[_borrowerID];

        // Add the borrower to the new lender if not already present
        if (bytes(newBorrower.borrowerID).length == 0) {
            newBorrower.borrowerID = _borrowerID;
            newBorrower.loanCount = 0;
            lenders[_newLenderID].borrowerIds.push(_borrowerID);
            lenders[_newLenderID].borrowerCount++;
        }

        // Assign a new loan ID for the new lender's borrower
        uint256 maxLoanID = 0;
        for (uint256 i = 0; i < newBorrower.loanIds.length; i++) {
            if (newBorrower.loanIds[i] > maxLoanID) {
                maxLoanID = newBorrower.loanIds[i];
            }
        }
        uint256 newLoanID = maxLoanID + 1;

        // Transfer the loan
        newBorrower.loans[newLoanID] = oldBorrower.loans[_loanID];
        newBorrower.loanIds.push(newLoanID);
        newBorrower.loanCount++;

        // Remove the loan from the old borrower
        for (uint256 i = 0; i < oldBorrower.loanIds.length; i++) {
            if (oldBorrower.loanIds[i] == _loanID) {
                oldBorrower.loanIds[i] = oldBorrower.loanIds[oldBorrower.loanIds.length - 1];
                oldBorrower.loanIds.pop();
                break;
            }
        }
        delete oldBorrower.loans[_loanID];
        oldBorrower.loanCount--;

        emit LoanTransferred(_oldLenderID, _newLenderID, _borrowerID, newLoanID);
    }

    // Function to delete a borrower
    function deleteBorrower(string memory _lenderID, string memory _borrowerID)
        external
        onlyAdmin
    {
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length > 0, "Lender does not exist");

        Borrower storage borrower = lender.borrowers[_borrowerID];
        require(
            bytes(borrower.borrowerID).length > 0,
            "Borrower does not exist"
        );

        delete lender.borrowers[_borrowerID];
        for (uint256 i = 0; i < lender.borrowerIds.length; i++) {
            if (
                keccak256(bytes(lender.borrowerIds[i])) ==
                keccak256(bytes(_borrowerID))
            ) {
                lender.borrowerIds[i] = lender.borrowerIds[
                    lender.borrowerIds.length - 1
                ];
                lender.borrowerIds.pop();
                break;
            }
        }
        lender.borrowerCount--;

        emit BorrowerDeleted(_lenderID, _borrowerID);
    }

    // Function to delete a loan
    function deleteLoan(
        string memory _lenderID,
        string memory _borrowerID,
        uint256 _loanID
    ) external onlyAdmin {
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length > 0, "Lender does not exist");

        Borrower storage borrower = lender.borrowers[_borrowerID];
        require(
            bytes(borrower.borrowerID).length > 0,
            "Borrower does not exist"
        );

        require(
            borrower.loans[_loanID].commonDetails.principalAmount > 0,
            "Loan does not exist"
        );
        delete borrower.loans[_loanID];

        for (uint256 i = 0; i < borrower.loanIds.length; i++) {
            if (borrower.loanIds[i] == _loanID) {
                borrower.loanIds[i] = borrower.loanIds[
                    borrower.loanIds.length - 1
                ];
                borrower.loanIds.pop();
                break;
            }
        }

        borrower.loanCount--;

        emit LoanDeleted(_lenderID, _borrowerID, _loanID);
    }

    // Function to delete a lender
    function deleteLender(string memory _lenderID) external onlyAdmin {
        // Check if the lender exists
        require(bytes(lenders[_lenderID].name).length > 0, "Lender does not exist");

        // Get the lender
        Lender storage lender = lenders[_lenderID];

        // Loop through and delete all borrowers under the lender
        for (uint256 i = 0; i < lender.borrowerIds.length; i++) {
            string memory borrowerID = lender.borrowerIds[i];
            delete lender.borrowers[borrowerID];
        }

        // Remove the lender's ID from the lenderIDs array
        for (uint256 i = 0; i < lenderIDs.length; i++) {
            if (
                keccak256(bytes(lenderIDs[i])) ==
                keccak256(bytes(_lenderID))
            ) {
                lenderIDs[i] = lenderIDs[lenderIDs.length - 1]; // Replace with the last element
                lenderIDs.pop(); // Remove the last element
                break;
            }
        }

        // Delete the lender from the mapping
        delete lenders[_lenderID];

        emit LenderDeleted(_lenderID);
    }


    // Function to get borrower data
    function getBorrowerData(string memory _lenderID, string memory _borrowerID)
        external
        view
        returns (string memory, uint256[] memory, uint256)
    {
        require(bytes(lenders[_lenderID].name).length > 0, "Lender does not exist");
        require(bytes(lenders[_lenderID].borrowers[_borrowerID].borrowerID).length > 0, "Borrower does not exist");

        Borrower storage borrower = lenders[_lenderID].borrowers[_borrowerID];
        return (borrower.borrowerID, borrower.loanIds, borrower.loanCount);
    }

    // Function to get loan details for a specific loan ID
    function getBorrowerLoanDetails(
        string memory _lenderID,
        string memory _borrowerID,
        uint256 _loanID
    ) external view returns (CommonAndInterestDetails memory, PaymentAndLateDetails memory, AdjustableInterestRateDetails memory) {
        require(bytes(lenders[_lenderID].name).length > 0, "Lender does not exist");
        require(bytes(lenders[_lenderID].borrowers[_borrowerID].borrowerID).length > 0, "Borrower does not exist");
        require(lenders[_lenderID].borrowers[_borrowerID].loans[_loanID].commonDetails.principalAmount > 0, "Loan does not exist");

        Loan storage loan = lenders[_lenderID].borrowers[_borrowerID].loans[_loanID];
        return (loan.commonDetails, loan.paymentDetails, loan.adjustableDetails);
    }

    // Function to get lender's data
    function getLenderData(string memory _lenderID)
        external
        view
        returns (string memory name, uint256 borrowerCount, string[] memory borrowerIds)
    {
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length > 0, "Lender does not exist");

        return (lender.name, lender.borrowerCount,lender.borrowerIds);
    }

    // Function to get all lender IDs
    function getAllLenderIDs() external view returns (string[] memory) {
        return lenderIDs;
    }

    // Function to get all borrower IDs for a specific lender
    function getAllBorrowerIDsUnderLender(string memory _lenderID)
        external
        view
        returns (string[] memory)
    {
        Lender storage lender = lenders[_lenderID];
        require(bytes(lender.name).length > 0, "Lender does not exist");

        return lender.borrowerIds;
    }

    // Contract Fund Management

    // Function to fund the contract with ETH
    function fundContract() external payable onlyAdmin {
        require(msg.value > 0, "Amount must be greater than zero");
        emit ContractFunded(Admin, msg.value);
    }

    // Function to withdraw funds from the contract
    function takeOutContractFunds(uint256 _amount) public onlyAdmin {
        require(_amount <= address(this).balance, "Insufficient funds");
        uint256 withdrawalAmount = _amount > 0
            ? _amount
            : address(this).balance;
        payable(Admin).transfer(withdrawalAmount);
        emit FundsWithdrawn(Admin, withdrawalAmount);
    }

    // Balance Check Functions

    // Function to get the admin's balance
    function getAdminBalance() public view onlyAdmin returns (uint256) {
        return Admin.balance;
    }

    // Function to check the balance of the smart contract
    function getBalanceOfSmartContract()
        public
        view
        onlyAdmin
        returns (uint256)
    {
        return address(this).balance;
    }

    // Fallback Functions

    // Fallback function to receive ETH
    receive() external payable {}

    // Fallback function to handle unexpected calls
    fallback() external payable {}
}
